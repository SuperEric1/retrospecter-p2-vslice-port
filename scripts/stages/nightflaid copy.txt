  var N_clr:StageProp;
  var N_top:StageProp;

  var N_rlt:StageProp;

  var N_lt2:StageProp;

  // Unbeatable elements

  var unbeatableLightningLeft:FlxSprite;
  var unbeatableLightningRight:FlxSprite;
  var needlesActive:Bool = false;

  // Light animation variables

  var lightRightBlinkTime:Float = 0;

  // Event tracking

  var dd:Bool = false;

  var animsWithOwnIdles:Array<String> = ['DAD', 'SONIC'];
  var groundedAnims:Array<String> = ['WHITTY', 'RUV', 'TABI', 'POKEMON'];
  var aerialAnims:Array<String> = ['GARCELLO', 'TRICKY', 'SHAGGY'];

  public function new()
  {
    super('nightflaid');
  }

  override function onCreate(event:ScriptEvent):Void
  {
    super.onCreate(event);
    if (hasCreatedStage) return;

    // Determine song variant
    var songName = PlayState.instance.currentSong.song;
    if (songName == 'Hexerpatient')
    {
      alt = '_green';
      colorr = '00FB3C';
    }
    else if (songName == 'Extiraging')
    {
      alt = '_cyan';
      colorr = '00FBCE';
    }

    // Create background layers
    N_hls = makeSprite('N_hls', 'nightflaid/highlights', bgX, bgY);
    N_drk = makeSprite('N_drk', 'nightflaid/darklayer' + alt, bgX, bgY);

    // Color layer (solid color)
    N_clr = FunkinSprite.create(bgX, bgY);
    N_clr.makeGraphic(Std.int(bgWidth), Std.int(bgHeight), FlxColor.WHITE);
    PlayState.instance.add(N_clr, 'N_clr');

    N_top = makeSprite('N_top', 'nightflaid/toplayer' + alt, bgX, bgY);

    // Scale color layer to match top layer
    N_clr.setGraphicSize(Std.int(N_top.width), Std.int(N_top.height));
    N_clr.updateHitbox();

    // Glow layer (foreground)
    N_glw = makeSprite('N_glw', 'nightflaid/glow' + alt, bgX, bgY);
    N_glw.zIndex = 200; // Above characters

    // Glow top
    N_glt = makeSprite('N_glt', 'nightflaid/glowTop' + alt, 140 + xAdjust, -1280 + yAdjust);
    N_glt.scale.set(2.65, 2.65);
    N_glt.updateHitbox();

    // Red lights
    N_rlt = makeSprite('N_rlt', 'nightflaid/redlights' + alt, -1375 + xAdjust, -555 + yAdjust);
    N_rlt.scale.set(2.65, 2.65);
    N_rlt.updateHitbox();

    // Light rays
    N_lt1 = makeSprite('N_lt1', 'nightflaid/lightray', -1340 + xAdjust, -765 + yAdjust);
    N_lt1.scale.set(2.65, 2.65);
    N_lt1.updateHitbox();

    N_lt2 = makeSprite('N_lt2', 'nightflaid/lightray', 2900 + xAdjust, -765 + yAdjust);
    N_lt2.scale.set(2.65, 2.65);
    N_lt2.flipX = true;
    N_lt2.updateHitbox();

    // Cover sprite (black screen overlay)
    coverSpr = FunkinSprite.create(-100, -100);
    coverSpr.makeGraphic(1, 1, FlxColor.BLACK);
    coverSpr.scrollFactor.set(0, 0);
    coverSpr.setGraphicSize(FlxG.width + 200, FlxG.height + 200);
    coverSpr.updateHitbox();
    coverSpr.alpha = 0;
    coverSpr.visible = false;
    coverSpr.zIndex = 1;
    PlayState.instance.add(coverSpr, 'coverSpr');

    // Unbeatable background
    unbeatableBG = makeSprite('unbeatableBG', 'nightflaid/unbeatableBG' + alt, -1000 + xAdjust, -900 + yAdjust);
    unbeatableBG.scrollFactor.set(0, 0);
    unbeatableBG.antialiasing = false;
    unbeatableBG.scale.set(150, 150);
    unbeatableBG.updateHitbox();
    unbeatableBG.alpha = 0;

    // White screen flash
    whiteScreen = FunkinSprite.create(1, 1);
    whiteScreen.makeGraphic(1, 1, FlxColor.WHITE);
    whiteScreen.scale.set(5000, 5000);
    whiteScreen.updateHitbox();
    whiteScreen.screenCenter();
    whiteScreen.alpha = 0.001;
    whiteScreen.zIndex = 300;
    PlayState.instance.add(whiteScreen, 'whiteScreen');

    hasCreatedStage = true;
    PlayState.instance.refresh();
  }

    {
      usedCheckPoint = true;
      if (getDad() != null) getDad().y = 350;
    }

  function makeSprite(name:String, imageName:String, x:Float, y:Float):StageProp
  {
    var sprite = FunkinSprite.create(x, y);
    sprite.loadTexture(imageName);
    PlayState.instance.add(sprite, name);
    return sprite;
  }

  function setupUnbeatableElements():Void
  {
    // Lightning sprites
    var leftPos:Float = 110.0 - 500.0;
    unbeatableLightningLeft = FunkinSprite.create(leftPos, -50);
    unbeatableLightningLeft.loadSparrow('nightflaid/ycbu_lightning');
    unbeatableLightningLeft.animation.addByPrefix('lightning', 'lightning', 24, true);
    unbeatableLightningLeft.animation.play('lightning', true);
    unbeatableLightningLeft.scrollFactor.set(0, 0);
    unbeatableLightningLeft.visible = false;
    unbeatableLightningLeft.active = false;
    PlayState.instance.add(unbeatableLightningLeft);

    var rightPos:Float = 1020.0 + 500.0;
    unbeatableLightningRight = FunkinSprite.create(rightPos, -50);
    unbeatableLightningRight.loadSparrow('nightflaid/ycbu_lightning');
    unbeatableLightningRight.animation.addByPrefix('lightning', 'lightning', 24, true);
    unbeatableLightningRight.animation.play('lightning', true);
    unbeatableLightningRight.scrollFactor.set(0, 0);
    unbeatableLightningRight.flipY = true;
    unbeatableLightningRight.visible = false;
    unbeatableLightningRight.active = false;
    PlayState.instance.add(unbeatableLightningRight);

    // Create needles
    for (i in 0...6)
    {
      var needleY:Float = 0.0;
      switch(i % 3)
      {
        case 1: needleY = 360.0;
        case 2: needleY = 720.0;
      }

      var needle = FunkinSprite.create(-1900, needleY);
      needle.scrollFactor.set(0, 0);
      needle.loadSparrow('nightflaid/needle');
      needle.animation.addByPrefix('needle', 'anim', 24, true);
      needle.animation.play('needle', true);
      needle.visible = false;
      needle.active = false;
      PlayState.instance.add(needle);

      if (i < 3)
      {
        unbeatableNeedlesLeft.push(needle);
      }
      else
      {
        needle.flipX = true;
        unbeatableNeedlesRight.push(needle);
      }
    }

    // Start unbeatable tween
    unbeatableTween();
  }

  function unbeatableTween():Void
  {
    if (unbeatableBG == null) return;

    FlxTween.tween(unbeatableBG, {
      x: FlxG.random.float(-1000.0, -1400.0),
      angle: FlxG.random.float(-25.0, 25.0)
    }, FlxG.random.float(2.5, 4.5), {
      ease: FlxEase.linear,
      onComplete: function(twn:FlxTween) {
        unbeatableTween();
      }
    });
  }

  function setCameraOnThings(top:Bool):Void
  {
    // TODO: Placeholder for camera switching
    // Would switch between normal view and "unbeatable" view
    needlesActive = top;

    if (top)
    {
      // Enable VCR/NTSC effects
      FlxTween.tween(unbeatableLightningLeft, {x: unbeatableLightningLeft.x + 500.0}, 0.75, {ease: FlxEase.quadOut});
      FlxTween.tween(unbeatableLightningRight, {x: unbeatableLightningRight.x - 500.0}, 0.75, {ease: FlxEase.quadOut});
    }
    else
    {
      // Disable effects
    }

    // Update needle visibility
    for (needle in unbeatableNeedlesLeft)
    {
      needle.active = needlesActive;
      needle.visible = needlesActive;
    }
    for (needle in unbeatableNeedlesRight)
    {
      needle.active = needlesActive;
      needle.visible = needlesActive;
    }
    unbeatableLightningLeft.visible = needlesActive;
    unbeatableLightningRight.visible = needlesActive;
    unbeatableLightningLeft.active = needlesActive;
    unbeatableLightningRight.active = needlesActive;
  }

  override function onUpdate(event:UpdateScriptEvent):Void
  {
    super.onUpdate(event);

    var elapsed = event.elapsed;

    // Light blinking logic - Left light
    lightLeftBlinkTime -= elapsed * 5.0;
    if (lightLeftBlinkTime < 0)
    {
      lightLeftBlinkCount = FlxG.random.float(2, 4);
      lightLeftBlinkTime = FlxG.random.float(12.0, 20.0);
    }
    else
    {
      if (lightLeftBlinkTime > lightLeftBlinkCount)
      {
        if (N_lt1 != null) N_lt1.visible = true;
      }
      else
      {
        if (N_lt1 != null) N_lt1.visible = (lightLeftBlinkTime % 1 < 0.5);
      }
    }

    // Light blinking logic - Right light
    lightRightBlinkTime -= elapsed * 5.0;
    if (lightRightBlinkTime < 0)
    {
      lightRightBlinkCount = FlxG.random.float(2, 4);
      lightRightBlinkTime = FlxG.random.float(12.0, 20.0);
    }
    else
    {
      if (lightRightBlinkTime > lightRightBlinkCount)
      {
        if (N_lt2 != null) N_lt2.visible = true;
      }
      else
      {
        if (N_lt2 != null) N_lt2.visible = (lightRightBlinkTime % 1 < 0.5);
      }
    }

    // Red light pulsing
    redLightTime += elapsed;
    if (N_rlt != null)
    {
      N_rlt.alpha = Math.sin(Math.PI * redLightTime) * 0.25 + 0.85;
    }

    // Update needles if active
    if (needlesActive)
    {
      for (needle in unbeatableNeedlesLeft)
      {
        needle.x = unbeatableLightningLeft.x - 80.0;
        needle.y -= 360 * elapsed;
        if (needle.y < -needle.height)
        {
          needle.y += (720.0 / 2.0) * 3.0;
        }
      }

      for (needle in unbeatableNeedlesRight)
      {
        needle.x = unbeatableLightningRight.x - 80.0;
        needle.y += 360 * elapsed;
        if (needle.y > FlxG.height)
        {
          needle.y -= (720.0 / 2.0) * 3.0;
        }
      }
    }
  }

  override function onBeatHit(event:SongTimeScriptEvent):Void
  {
    super.onBeatHit(event);

    var curBeat = event.beat;

    // Zoom on specific beats
    for (zoomBeat in zoomBeats)
    {
      if (curBeat == zoomBeat)
      {
        var crochet = (Conductor.instance.stepLengthMs * 4) / 1000;

        FlxTween.tween(PlayState.instance.camGame, {zoom: PlayState.instance.camGame.zoom + 0.1, angle: -4.5}, crochet, {
          ease: FlxEase.expoIn,
          onComplete: function(twn:FlxTween) {
            FlxTween.tween(PlayState.instance.camGame, {angle: 0}, crochet, {ease: FlxEase.expoOut});
          }
        });

        FlxTween.tween(PlayState.instance.camHUD, {zoom: 1.1, angle: 4}, crochet, {
          ease: FlxEase.expoIn,
          onComplete: function(twn:FlxTween) {
            FlxTween.tween(PlayState.instance.camHUD, {angle: 0}, crochet, {ease: FlxEase.expoOut});
          }
        });
      }
    }
  }

  override function onStepHit(event:SongTimeScriptEvent):Void
  {
    super.onStepHit(event);

    var curStep = event.step;

    // Step-based events
    if (curStep >= 3512 && !finalFadeout && PlayState.instance.isStoryMode)
    {
      finalFadeout = true;
      if (coverSpr != null)
      {
        coverSpr.alpha = 0;
        coverSpr.visible = true;
        FlxTween.tween(coverSpr, {alpha: 1}, 0.5);
      }
    }

    if (curStep >= 64 && !sq && !usedCheckPoint)
    {
      sq = true;
      // triggerNightflaidColorTween('CCCCCC', 0.15);
    }

    if (curStep >= 2944 && !qqs)
    {
      qqs = true;
      // triggerNightflaidLightings(true);
    }

    if (curStep >= 3456 && !qqas)
    {
      qqas = true;
      // triggerNightflaidLightings(false);
      // triggerNightflaidColorTween('0d0d0d', 0.15);
    }

    if (curStep >= 2304 && !qweqe)
    {
      // triggerNightflaidColorTweenRightOnly('0d0d0d', 0.3);
      // triggerNightflaidColorTweenLeftOnly('0d0d0d', 0.3);
      qweqe = true;
      qe = false;
    }

    if (curStep >= 2432 && !qqqqq)
    {
      // triggerNightflaidColorTweenLeftOnly(colorr, 0.3);
      qqqqq = true;
      qe = true;
    }
  }

  function handleCutsceneAnimation(animName:String):Void
  {
    var dad = getDad();
    if (dad == null) return;

    var groundedOff = 250;
    var crochet = Conductor.instance.stepLengthMs * 4;

    switch (animName)
    {
      case 'descend':
        FlxTween.cancelTweensOf(dad);
        FlxTween.tween(dad, {y: dad.y + groundedOff}, crochet / 400, {ease: FlxEase.quadInOut});

      case 'ascend':
        FlxTween.cancelTweensOf(dad);
        FlxTween.tween(dad, {y: dad.y - groundedOff}, crochet / 400, {ease: FlxEase.quadInOut});

      case 'final':
        // TODO: Fade out HUD
    }
  }

  function handleOpponentCharacterChange(character:String):Void
  {
    var dad = getDad();
    if (dad == null) return;

    if (character == '80sNightflaid')
    {
      // Special setup for 80s Nightflaid character
      if (dad.animation != null)
      {
        dad.playAnimation('singLEFT', true);
      }

      if (coverSpr != null)
      {
        FlxTween.cancelTweensOf(coverSpr);
        coverSpr.alpha = 1;
      }

      var posY = 200;
      dad.y = posY + FlxG.height + 100;
      FlxTween.tween(dad, {y: posY}, 0.75, {ease: FlxEase.circOut});

      if (unbeatableBG != null)
      {
        FlxTween.tween(unbeatableBG, {alpha: 1}, 0.75, {ease: FlxEase.circOut});
      }

      setCameraOnThings(true);

      // TODO: Update health bar colors
    }
  }

  function handleMidsongCutscene():Void
  {
    var dad = getDad();
    if (dad == null) return;

    // Transition dad out
    FlxTween.tween(dad, {y: dad.y + FlxG.height + 100}, 0.5, {ease: FlxEase.sineIn});

    if (unbeatableBG != null)
    {
      FlxTween.tween(unbeatableBG, {alpha: 0}, 0.5, {ease: FlxEase.circOut});
    }

    // Pre-transition effects
    if (unbeatableLightningLeft != null)
    {
      FlxTween.tween(unbeatableLightningLeft, {x: unbeatableLightningLeft.x - 500.0}, 0.5, {ease: FlxEase.quadIn});
    }
    if (unbeatableLightningRight != null)
    {
      FlxTween.tween(unbeatableLightningRight, {x: unbeatableLightningRight.x + 500.0}, 0.5, {ease: FlxEase.quadIn});
    }

    // TODO: Update health bar colors to white

    // After animation completes, switch character and reset
    new FlxTimer().start(0.5, function(tmr:FlxTimer) {
      if (coverSpr != null)
      {
        coverSpr.alpha = 1;
        FlxTween.tween(coverSpr, {alpha: 0}, 0.5);
      }

      setCameraOnThings(false);

      // TODO: Reset color tween to dark
    });
  }

  override function addCharacter(character:BaseCharacter, charType:CharacterType):Void
  {
    super.addCharacter(character, charType);

    switch (charType)
    {
      case CharacterType.GF:
        character.visible = false;
      default:
    }
  }

  override function onDestroy(event:ScriptEvent):Void
  {
    super.onDestroy(event);
    hasCreatedStage = false;

    // Cleanup arrays
    unbeatableNeedlesLeft = [];
    unbeatableNeedlesRight = [];
  }
